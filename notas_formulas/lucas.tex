	$\dbinom{n}{m} \equiv \prod_i \dbinom{n_i}{m_i} \pmod p$ \quad ($n_i$, $m_i$: d\'igitos de $n$, $m$ en base $p$, $p$ primo) \\
	Si alg\'un $m_i > n_i$ el resultado es $0$. \quad $\dbinom{n}{m}$ es impar $\iff$ \texttt{(n \& m) == m} \\
\end{tabular}
